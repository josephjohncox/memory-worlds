  umask 077
  kubectl --context YOUR_CONTEXT -n {{ .Release.Namespace }} exec {{ include "memory-worlds.fullname" . }}-0 -- memory-worlds init > /private/operator/bootstrap.json

Forward the authenticated API to loopback:
  kubectl --context YOUR_CONTEXT -n {{ .Release.Namespace }} port-forward --address=127.0.0.1 service/{{ include "memory-worlds.fullname" . }} 8787:8787

Run read-only deployment checks:
  helm --kube-context YOUR_CONTEXT -n {{ .Release.Namespace }} test {{ .Release.Name }}

unless explicitly permitted. Existing PVCs and Secrets are not deleted on uninstall.
